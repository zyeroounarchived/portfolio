# Objective: Maximize profit given resource constraints
library(lpSolve)

# Objective function coefficients (Profit per unit)
obj_coeff <- c(40, 30) 

# Constraint matrix
const_mat <- matrix(c(2, 1, 
                      1, 2,
                      1, 1), nrow=3, byrow=TRUE)

# Constraint directions and RHS values
const_dir <- c("<=", "<=", "<=")
const_rhs <- c(100, 80, 50)

# Solve
solution <- lp("max", obj_coeff, const_mat, const_dir, const_rhs)
print(solution$solution) # Output optimal quantities
